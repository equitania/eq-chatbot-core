      class RealtimeClientError(Exception): pass  # base
      class RealtimeConnectionError(RealtimeClientError): pass  # failed to connect / network error
      class RealtimeClosedError(RealtimeClientError):
          __init__(self, message: str, code: int | None = None, retriable: bool = False)
          fields: self.code, self.retriable
          Docstring: "code: WebSocket close code (1000=normal, 1006=abnormal). retriable: True when code != 1000."
      class RealtimeRateLimitError(RealtimeClientError):
          __init__(self, message: str, retry_after: float | None = None)
          field: self.retry_after
      class RealtimeProtocolError(RealtimeClientError): pass  # malformed/unexpected frame

    Class BaseRealtimeWebsocketClient(ABC):
      Module docstring: "WebSocket base class for realtime providers. Handles connect/send/recv/close lifecycle, connection-leak protection, exponential backoff reconnect, and error normalization across websockets 13.x–16.x."

      __init__(self, url: str, headers: dict[str, str] | None = None) -> None:
        self._url = url
        self._headers = headers or {}
        self._ws: Any = None

      @property is_connected(self) -> bool:
        return self._ws is not None and not getattr(self._ws, "closed", True)

      @abstractmethod async def _on_connected(self) -> None: ...
        Docstring: "Called after WebSocket handshake completes. Raise to abort — connect() will call close()."

      @abstractmethod async def _on_message(self, raw: str) -> None: ...
        Docstring: "Called for each received text frame."

      @abstractmethod def _connection_error_endpoint(self) -> str:
        Docstring: "Return a REDACTED URL string safe for error messages and logs. Override to strip API keys. Default returns self._url unchanged (safe only for providers that don't embed secrets in URL)."
        Default implementation: return self._url

      async def connect(self) -> None:
        If already connected (self.is_connected), return immediately.
        Attempt websockets.connect(self._url, additional_headers=self._headers) first; on TypeError try extra_headers (cross-version fallback per GlassAgents pattern).
        Wrap in try/except for ws_exceptions.WebSocketException; map HTTP 429 (via cross-version status_code introspection: getattr(getattr(exc, "response", None), "status_code", None) or getattr(exc, "status_code", None)) to RealtimeRateLimitError; map other connection failures to RealtimeConnectionError.
        After self._ws is assigned, call await self._on_connected() inside try/finally — on any exception: await self.close() then re-raise (PITFALL-04 fix).

      async def close(self) -> None:
        Idempotent. If self._ws is None, return. Call await self._ws.close() inside try/except Exception (swallow close errors), then set self._ws = None.

      async def send_json(self, data: dict[str, Any]) -> None:
        Import json. Serialize data with json.dumps and send as text frame. Raise RealtimeConnectionError if not connected.

      async def recv_json(self) -> dict[str, Any]:
        Call await self._ws.recv(). Import json. Parse with json.loads. Wrap ws_exceptions.ConnectionClosed to RealtimeClosedError:
          code = getattr(exc, "code", None) or getattr(getattr(exc, "rcvd", None), "code", None)
          retriable = code != 1000
          raise RealtimeClosedError("WebSocket closed", code=code, retriable=retriable) from exc

      async def connect_with_backoff(self, max_attempts: int = 5, base_delay_s: float = 1.0, max_delay_s: float = 30.0) -> None:
        Implements truncated exponential backoff with jitter per RESEARCH.md Pattern 3:
          for attempt in range(max_attempts):
            try: await self.connect(); return
            except RealtimeConnectionError as exc:
              if attempt < max_attempts - 1:
                delay = min(base_delay_s * (2 ** attempt) + random.uniform(0, 1), max_delay_s)
                _logger.warning("Realtime connect attempt %d/%d failed, retrying in %.1fs: %s", attempt + 1, max_attempts, delay, exc)
                await asyncio.sleep(delay)
              else: last_exc = exc
          raise RealtimeConnectionError(f"Failed to connect after {max_attempts} attempts") from last_exc

      async def __aenter__(self) -> "BaseRealtimeWebsocketClient":
        await self.connect(); return self

      async def __aexit__(self, *_: Any) -> None:
        await self.close()

    __all__ = ["BaseRealtimeWebsocketClient", "RealtimeClientError", "RealtimeConnectionError", "RealtimeClosedError", "RealtimeRateLimitError", "RealtimeProtocolError"]

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Library → WebSocket server | BaseRealtimeWebsocketClient opens TCP+TLS connection to provider endpoint. API keys may be in headers or URL (Gemini). This is the primary new trust boundary for the entire realtime feature. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Information Disclosure | API key in WebSocket URL (Gemini pattern) | mitigate | _connection_error_endpoint() abstract hook forces each concrete provider to override and return a redacted URL for logging; base default returns raw URL (acceptable for OpenAI which uses headers, not URL query params) |
| T-03-02 | Information Disclosure | API key in RealtimeConnectionError message | mitigate | connect() error path uses self._connection_error_endpoint() for the message, not the raw URL or headers |
| T-03-03 | Denial of Service | connect_with_backoff without max_delay_s cap | mitigate | max_delay_s=30.0 default caps delay; min() ensures no unbounded growth |
| T-03-04 | Tampering | Untrusted WebSocket server returning crafted JSON | accept | recv_json() returns raw dict; consumer is responsible for schema validation on received frames; library does not re-interpret frame content |
| T-03-05 | Spoofing | TLS verification bypass | mitigate | websockets.connect() uses Python's default ssl context (certificate verification enabled); no ssl=False or custom context passed; TLS is websockets-library responsibility under Python's default ssl module |
| T-03-06 | Elevation of Privilege | _on_connected() raises and leaves TCP socket open (PITFALL-04) | mitigate | connect() wraps _on_connected() in try/finally; calls await self.close() on failure before re-raising; prevents connection leak |
| T-03-SC | Tampering | websockets package install | mitigate | Human-verify checkpoint (Task 1) gates install; package audit in RESEARCH.md confirms [ASSUMED-OK] |
</threat_model>
