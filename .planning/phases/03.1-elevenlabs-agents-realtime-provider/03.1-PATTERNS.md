def __init__(self, config: OpenAIRealtimeConfig, *, trace_events: bool = False) -> None:
    if not config.api_key.strip():

    if not config.model.strip():
        raise ValueError("OpenAIRealtimeConfig.model must be non-empty. ...")

**ElevenLabs delta:** Validate `api_key` AND `agent_id`. URL contains NO key. For public agents, `headers={}`. The `xi-api-key` is only used in a REST call to obtain a signed URL — it never goes in the WebSocket URL. The base class receives empty headers.

```python
def __init__(self, config: ElevenLabsRealtimeConfig, *, trace_events: bool = False) -> None:
    if not config.api_key.strip():
        raise ValueError("ElevenLabsRealtimeConfig.api_key must be non-empty.")
    if not config.agent_id.strip():
        raise ValueError("ElevenLabsRealtimeConfig.agent_id must be non-empty.")

@staticmethod
def _redact_sensitive_url(url: str) -> str:
    parts = urlsplit(url)
    params = parse_qsl(parts.query, keep_blank_values=True)
    redacted = [(k, "[REDACTED]" if k == "key" else v) for k, v in params]
    new_query = urlencode(redacted, quote_via=quote, safe="[]")
    return urlunsplit(parts._replace(query=new_query))
```

**ElevenLabs pattern:** The `xi-api-key` never appears in the WebSocket URL (it is used only in the REST signed-URL flow). The `agent_id` is not a secret and may appear in error logs. Use the **simple pattern** from `openai.py`:

async def initialize_session(
    self,
    *,
    instructions: str | None = None,
    voice: str | None = None,
    tools: list[Any] | None = None,
) -> None:
    payload: dict[str, Any] = {
        "type": "conversation_initiation_client_data",
        "conversation_config_override": {},
    }
    resolved_instructions = instructions or self._config.instructions
    if resolved_instructions:
        payload["conversation_config_override"]["agent"] = {
            "prompt": {"prompt": resolved_instructions}

    resolved_voice = voice or self._config.voice
    if resolved_voice:
        payload["conversation_config_override"]["tts"] = {"voice_id": resolved_voice}
    if tools:
        # ElevenLabs client tool registration — include in override payload
        payload["conversation_config_override"]["agent"]["custom_llm"] = {
            "tools": self._normalize_tools(tools)
        }
    await self.send_json(payload)

**ElevenLabs dispatch table (from RESEARCH.md Open Point 3 — verified event names):**

```python
def _to_normalized_runtime_event(
    self, event: dict[str, Any]
) -> NormalizedRealtimeEventFull | None:
    """Map ElevenLabs wire events to NormalizedRealtimeEventTypes.

    Returns None for events that should NOT be yielded (ping — handled internally).
    Caller (iter_normalized_events) must filter None results.
    """
    event_type = event.get("type")
    if not isinstance(event_type, str):
        return {"type": NormalizedRealtimeEventTypes.UNHANDLED,
                "payload": event, "source": "missing_type", "raw": event}

    if event_type == "conversation_initiation_metadata":
        return {"type": NormalizedRealtimeEventTypes.SESSION_READY,
                "payload": event, "source": event_type, "raw": event}

    if event_type == "audio":
        audio_b64 = event.get("audio_event", {}).get("audio_base_64", "")
        return {"type": NormalizedRealtimeEventTypes.RESPONSE_AUDIO_DELTA,
                "payload": {"audio": audio_b64}, "source": event_type, "raw": event}

    if event_type == "agent_response":
        # Text-only — audio comes via "audio" events. Map to RESPONSE_DONE as best proxy.
        # TODO: "agent_response_complete" may become the canonical RESPONSE_DONE signal
        # once ElevenLabs SDK adds it to the wire dispatch (added to API schema April 2026).
        return {"type": NormalizedRealtimeEventTypes.RESPONSE_DONE,
                "payload": event, "source": event_type, "raw": event}

    if event_type == "agent_response_complete":
        # Forward-compatible: not in current SDK wire dispatch (v2.49.1) but safe to handle.
        return {"type": NormalizedRealtimeEventTypes.RESPONSE_DONE,
                "payload": event, "source": event_type, "raw": event}

    if event_type == "user_transcript":
        return {"type": NormalizedRealtimeEventTypes.INPUT_AUDIO_COMMITTED,
                "payload": event, "source": event_type, "raw": event}

    if event_type == "interruption":
        # User interrupted the agent (server VAD detected speech).
        return {"type": NormalizedRealtimeEventTypes.INPUT_SPEECH_STARTED,
                "payload": event, "source": event_type, "raw": event}

    if event_type == "ping":
        # Keepalive — respond with pong and do NOT yield a normalized event.
        # Caller filters None returns.
        event_id = event.get("ping_event", {}).get("event_id")
        # NOTE: send_json is async; the caller (iter_normalized_events) must await this.
        # Pattern: return a sentinel or handle inline — see iter_normalized_events below.
        return None  # sentinel — caller handles pong send

    if event_type == "client_tool_call":
        tool_info = event.get("client_tool_call", {})
        return {
            "type": NormalizedRealtimeEventTypes.TOOL_CALL_COMPLETED,
            "payload": {
                "call_id": tool_info.get("tool_call_id"),
                "name": tool_info.get("tool_name"),
                "arguments": tool_info.get("parameters", "{}"),
            },
            "source": event_type,
            "raw": event,
        }

    if event_type == "agent_response_correction":
        return {"type": NormalizedRealtimeEventTypes.UNHANDLED,
                "payload": event, "source": event_type, "raw": event}

    return {"type": NormalizedRealtimeEventTypes.UNHANDLED,
            "payload": event, "source": event_type or "unknown", "raw": event}
```

async def test_elevenlabs_session_ready_and_disconnect() -> None:
    """QUAL-03: Connect to real ElevenLabs agent, receive SESSION_READY, disconnect cleanly.

    Flow:
        1. Open WebSocket connection via async context manager
        2. Iterate normalized events until SESSION_READY is received (server sends it first)
        3. Optionally call initialize_session() to send conversation_initiation_client_data
        4. Send 100ms of PCM16 silence (1600 bytes at 16kHz mono)

    Note: No commit_client_turn() — ElevenLabs uses server-side VAD.
    Note: initialize_session() is optional here (no override needed for the smoke test).

        agent_id=os.environ["ELEVENLABS_AGENT_ID"],
        base_url=os.getenv("ELEVENLABS_BASE_URL", "wss://api.elevenlabs.io"),

    async with ElevenLabsRealtimeClient(config) as client:
